# Restock Notification Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Go HTTP restock notification service with SQLite-backed transactional outbox delivery.

**Architecture:** The service uses a small layered Go design: HTTP handlers call an application service, the service owns SQLite transactions, and a background worker processes durable notification jobs. SQLite uniqueness constraints and idempotent state transitions provide the no-miss and no-duplicate behavior.

**Tech Stack:** Go 1.26, `database/sql`, `modernc.org/sqlite`, `net/http`, `httptest`, SQLite.

## Global Constraints

- Build from scratch in the current repository.
- Use SQLite as the only database.
- Expose an HTTP JSON API plus an in-process background worker.
- Use a transactional outbox for restock notification jobs.
- Registrations are one-shot and complete after successful notification.
- Muted notifications are deferred until the mute window ends.
- Implementation must follow TDD: write each failing test before production code.
- Use real SQLite in tests; avoid mocks except for a small notifier interface.
- Keep the implementation interview-sized; no authentication, real email, distributed workers, or UI.

```go
package store

import (
	"context"
	"database/sql"
	"testing"

	_ "modernc.org/sqlite"
)

func Open(path string) (*sql.DB, error) {
	db, err := sql.Open("sqlite", path)
	if err != nil {
		return nil, err
	}
	db.SetMaxOpenConns(1)
	if err := Migrate(context.Background(), db); err != nil {
		_ = db.Close()
		return nil, err
	}
	return db, nil
}

func OpenTestDB(t testing.TB) *sql.DB {
	t.Helper()
	db, err := Open("file::memory:?cache=shared")
	if err != nil {
		t.Fatalf("open test db: %v", err)
	}
	t.Cleanup(func() {
		if err := db.Close(); err != nil {
			t.Fatalf("close test db: %v", err)
		}
	})
	return db
}

func Migrate(ctx context.Context, db *sql.DB) error {
	_, err := db.ExecContext(ctx, schema)
	return err
}

const schema = `
PRAGMA foreign_keys = ON;

CREATE TABLE IF NOT EXISTS products (
	id INTEGER PRIMARY KEY AUTOINCREMENT,
	sku TEXT NOT NULL UNIQUE,
	name TEXT NOT NULL,
	stock INTEGER NOT NULL CHECK (stock >= 0),
	created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
	updated_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS customers (
	id INTEGER PRIMARY KEY AUTOINCREMENT,
	email TEXT NOT NULL UNIQUE,
	mute_start_minute INTEGER,
	mute_end_minute INTEGER,
	created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
	updated_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
	CHECK (
		(mute_start_minute IS NULL AND mute_end_minute IS NULL)
		OR
		(mute_start_minute BETWEEN 0 AND 1439 AND mute_end_minute BETWEEN 0 AND 1439)
	)
);

CREATE TABLE IF NOT EXISTS restock_registrations (
	id INTEGER PRIMARY KEY AUTOINCREMENT,
	product_id INTEGER NOT NULL REFERENCES products(id),
	customer_id INTEGER NOT NULL REFERENCES customers(id),
	status TEXT NOT NULL CHECK (status IN ('active', 'queued', 'notified')),
	created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
	updated_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE UNIQUE INDEX IF NOT EXISTS uniq_live_registration
ON restock_registrations(product_id, customer_id)
WHERE status IN ('active', 'queued');

CREATE TABLE IF NOT EXISTS restock_events (
	id INTEGER PRIMARY KEY AUTOINCREMENT,
	product_id INTEGER NOT NULL REFERENCES products(id),
	previous_stock INTEGER NOT NULL,
	new_stock INTEGER NOT NULL,
	created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS notification_jobs (
	id INTEGER PRIMARY KEY AUTOINCREMENT,
	registration_id INTEGER NOT NULL UNIQUE REFERENCES restock_registrations(id),
	restock_event_id INTEGER NOT NULL REFERENCES restock_events(id),
	customer_id INTEGER NOT NULL REFERENCES customers(id),
	product_id INTEGER NOT NULL REFERENCES products(id),
	status TEXT NOT NULL CHECK (status IN ('pending', 'processing', 'sent', 'failed')),
	available_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
	attempt_count INTEGER NOT NULL DEFAULT 0,
	last_error TEXT,
	created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
	updated_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX IF NOT EXISTS idx_notification_jobs_due
ON notification_jobs(status, available_at, id);

CREATE TABLE IF NOT EXISTS notification_deliveries (
	id INTEGER PRIMARY KEY AUTOINCREMENT,
	job_id INTEGER NOT NULL UNIQUE REFERENCES notification_jobs(id),
	customer_id INTEGER NOT NULL REFERENCES customers(id),
	product_id INTEGER NOT NULL REFERENCES products(id),
	message TEXT NOT NULL,
	sent_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);
`
```

```go
func (s *Service) CreateProduct(ctx context.Context, in CreateProductInput) (Product, error) {
	if strings.TrimSpace(in.SKU) == "" || strings.TrimSpace(in.Name) == "" || in.Stock < 0 {
		return Product{}, ErrInvalidInput
	}
	res, err := s.db.ExecContext(ctx, `
		INSERT INTO products (sku, name, stock, updated_at)
		VALUES (?, ?, ?, CURRENT_TIMESTAMP)`, in.SKU, in.Name, in.Stock)
	if err != nil {
		return Product{}, err
	}
	id, err := res.LastInsertId()
	if err != nil {
		return Product{}, err
	}
	return Product{ID: id, SKU: in.SKU, Name: in.Name, Stock: in.Stock}, nil
}
```

```go
func (s *Service) RegisterRestock(ctx context.Context, in RegisterRestockInput) (Registration, error) {
	if in.ProductID == 0 || strings.TrimSpace(in.Email) == "" {
		return Registration{}, ErrInvalidInput
	}
	tx, err := s.db.BeginTx(ctx, nil)
	if err != nil {
		return Registration{}, err
	}
	defer tx.Rollback()

	var stock int
	if err := tx.QueryRowContext(ctx, `SELECT stock FROM products WHERE id = ?`, in.ProductID).Scan(&stock); err != nil {
		if errors.Is(err, sql.ErrNoRows) {
			return Registration{}, ErrNotFound
		}
		return Registration{}, err
	}
	if stock > 0 {
		return Registration{}, ErrProductInStock
	}

	customerID, err := upsertCustomer(ctx, tx, in.Email)
	if err != nil {
		return Registration{}, err
	}
	res, err := tx.ExecContext(ctx, `
		INSERT INTO restock_registrations (product_id, customer_id, status, updated_at)
		VALUES (?, ?, 'active', CURRENT_TIMESTAMP)`, in.ProductID, customerID)
	if err != nil {
		return Registration{}, err
	}
	registrationID, err := res.LastInsertId()
	if err != nil {
		return Registration{}, err
	}
	if err := tx.Commit(); err != nil {
		return Registration{}, err
	}
	return Registration{ID: registrationID, ProductID: in.ProductID, CustomerID: customerID, Status: "active"}, nil
}
```

```go
func (s *Service) Replenish(ctx context.Context, productID int64, quantity int) (ReplenishResult, error) {
	if productID == 0 || quantity <= 0 {
		return ReplenishResult{}, ErrInvalidInput
	}
	tx, err := s.db.BeginTx(ctx, nil)
	if err != nil {
		return ReplenishResult{}, err
	}
	defer tx.Rollback()

	var previous int
	if err := tx.QueryRowContext(ctx, `SELECT stock FROM products WHERE id = ?`, productID).Scan(&previous); err != nil {
		if errors.Is(err, sql.ErrNoRows) {
			return ReplenishResult{}, ErrNotFound
		}
		return ReplenishResult{}, err
	}
	newStock := previous + quantity
	if _, err := tx.ExecContext(ctx, `
		UPDATE products SET stock = ?, updated_at = CURRENT_TIMESTAMP WHERE id = ?`, newStock, productID); err != nil {
		return ReplenishResult{}, err
	}

	result := ReplenishResult{ProductID: productID, PreviousStock: previous, NewStock: newStock}
	if previous == 0 && newStock > 0 {
		eventID, created, err := createRestockEventAndJobs(ctx, tx, productID, previous, newStock)
		if err != nil {
			return ReplenishResult{}, err
		}
		result.RestockEventID = eventID
		result.CreatedJobs = created
	}
	if err := tx.Commit(); err != nil {
		return ReplenishResult{}, err
	}
	return result, nil
}
```

```go
	now := w.clock().UTC()
	rows, err := w.db.QueryContext(ctx, `
		SELECT j.id, j.customer_id, j.product_id, c.email, c.mute_start_minute, c.mute_end_minute
		FROM notification_jobs j
		JOIN customers c ON c.id = j.customer_id
		WHERE j.status = 'pending' AND j.available_at <= ?
		ORDER BY j.id
		LIMIT ?`, now.Format(time.RFC3339), limit)
	if err != nil {
		return WorkerResult{}, err
	}
	defer rows.Close()

	type job struct {
		id, customerID, productID int64
		email                    string
		start, end               sql.NullInt64
	}
	var jobs []job
	for rows.Next() {
		var j job
		if err := rows.Scan(&j.id, &j.customerID, &j.productID, &j.email, &j.start, &j.end); err != nil {
			return WorkerResult{}, err
		}
		jobs = append(jobs, j)
	}
	if err := rows.Err(); err != nil {
		return WorkerResult{}, err
	}

	result := WorkerResult{Claimed: len(jobs)}
	for _, j := range jobs {
		if err := w.processJob(ctx, j.id, j.customerID, j.productID, j.email, nullMinute(j.start), nullMinute(j.end), now, &result); err != nil {
			return result, err
		}
	}
	return result, nil
}
```

```go
func IsMutedAt(start *int, end *int, now time.Time) bool {
	if start == nil || end == nil {
		return false
	}
	minute := now.Hour()*60 + now.Minute()
	if *start == *end {
		return false
	}
	if *start < *end {
		return minute >= *start && minute < *end
	}
	return minute >= *start || minute < *end
}

func MuteEndAfter(start int, end int, now time.Time) time.Time {
	base := time.Date(now.Year(), now.Month(), now.Day(), end/60, end%60, 0, 0, now.Location())
	if start > end && (now.Hour()*60+now.Minute()) >= start {
		return base.Add(24 * time.Hour)
	}
	if !base.After(now) {
		return base.Add(24 * time.Hour)
	}
	return base
}
```

```go
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	_ = json.NewEncoder(w).Encode(body)
}

func writeError(w http.ResponseWriter, err error) {
	status := http.StatusInternalServerError
	switch {
	case errors.Is(err, app.ErrInvalidInput):

	case errors.Is(err, app.ErrProductInStock):
		status = http.StatusConflict
	case errors.Is(err, app.ErrNotFound):

	}

}
```

```go
package main

import (
	"context"
	"log"
	"net/http"
	"os"
	"os/signal"
	"time"

	"github.com/hanfezh/restock-notification/internal/app"
	"github.com/hanfezh/restock-notification/internal/httpapi"
	"github.com/hanfezh/restock-notification/internal/store"
)

func main() {
	dbPath := getenv("RESTOCK_DB_PATH", "restock.db")
	addr := getenv("RESTOCK_ADDR", ":8080")

	db, err := store.Open(dbPath)
	if err != nil {
		log.Fatalf("open db: %v", err)
	}
	defer db.Close()

	clock := func() time.Time { return time.Now().UTC() }
	service := app.NewService(db, clock)
	worker := app.NewWorker(db, clock)
	handler := httpapi.NewServer(service, worker)

	server := &http.Server{Addr: addr, Handler: handler}
	go func() {
		ticker := time.NewTicker(time.Second)
		defer ticker.Stop()
		for range ticker.C {
			if _, err := worker.ProcessBatch(context.Background(), 25); err != nil {
				log.Printf("process notifications: %v", err)
			}
		}
	}()

	go func() {
		log.Printf("listening on %s", addr)
		if err := server.ListenAndServe(); err != nil && err != http.ErrServerClosed {
			log.Fatalf("serve: %v", err)
		}
	}()

	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt)
	defer stop()
	<-ctx.Done()

	shutdownCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	if err := server.Shutdown(shutdownCtx); err != nil {
		log.Printf("shutdown: %v", err)
	}
}

func getenv(key string, fallback string) string {
	if value := os.Getenv(key); value != "" {
		return value
	}
	return fallback
}
```
